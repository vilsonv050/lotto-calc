function gammaQ(a,x){if(x<=0)return 1;
if(x<a+1)return 1-gammaSeriesP(a,x);
return gammaCFQ(a,x);
}
function gammaSeriesP(a,x){var maxIter=200,eps=3e-7;
if(x===0)return 0;
var ap=a,sum=1 / a,del=sum;
for(var n=1;
n<=maxIter;
n++){ap+=1;
del *=x / ap;
sum+=del;
if(Math.abs(del)<Math.abs(sum)* eps)break;
}return sum * Math.exp(-x+a * Math.log(x)-lnGamma(a));
}
function gammaCFQ(a,x){var maxIter=200,eps=3e-7,tiny=1e-30;
var b=x+1-a,c=1 / tiny,d=1 / b,h=d;
for(var i=1;
i<=maxIter;
i++){var an=-i *(i-a);
b+=2;
d=an * d+b;
if(Math.abs(d)<tiny)d=tiny;
c=b+an / c;
if(Math.abs(c)<tiny)c=tiny;
d=1 / d;
var del=d * c;
h *=del;
if(Math.abs(del-1)<eps)break;
}return Math.exp(-x+a * Math.log(x)-lnGamma(a))* h;
}

function setupCanvas(canvas){var dpr=window.devicePixelRatio||1;

function drawBarChart(canvas,data,opts){var s=setupCanvas(canvas);
var ctx=s.ctx,W=s.w,H=s.h;
var pad=opts.padding||{top:10,right:10,bottom:25,left:40};
var chartW=W-pad.left-pad.right;
var chartH=H-pad.top-pad.bottom;
var n=data.length;
if(n===0)return;
var maxVal=opts.maxVal||0;
for(var i=0;
i<n;
i++){if(data[i].value>maxVal)maxVal=data[i].value;
}maxVal=maxVal * 1.15;
var barW=opts.barWidth||Math.max(2,(chartW / n)-1);
var gap=(chartW-barW * n)/(n+1);
ctx.fillStyle="#0d0d1a";
ctx.fillRect(0,0,W,H);
ctx.strokeStyle="#222";
ctx.lineWidth=0.5;
for(var i=0;
i<=4;
i++){var yy=pad.top+chartH-(chartH / 4 * i);
ctx.beginPath();
ctx.moveTo(pad.left,yy);
ctx.lineTo(W-pad.right,yy);
ctx.stroke();
}var fs=opts.fontSize||10;
ctx.textAlign="center";
for(var i=0;
i<n;
i++){var d=data[i];
var x=pad.left+gap+i *(barW+gap);
var barH=maxVal>0?(d.value / maxVal)* chartH:0;
var y=pad.top+chartH-barH;
ctx.fillStyle=d.color||"#4CAF50";
ctx.fillRect(x,y,barW,barH);
if(opts.showValues!==false&&barW>=8){ctx.fillStyle="#ccc";
ctx.font=(fs-1)+"px monospace";
ctx.fillText(d.value,x+barW / 2,y-2);
}if(opts.showLabels!==false&&barW>=6){ctx.fillStyle="#888";
ctx.font=fs+"px monospace";
ctx.fillText(d.label,x+barW / 2,H-4);
}}}
function drawLineChart(canvas,points,opts){var s=setupCanvas(canvas);
var ctx=s.ctx,W=s.w,H=s.h;
var pad=opts.padding||{top:10,right:10,bottom:25,left:45};

i<points.length;
i++){vals.push(typeof points[i]==="number"?points[i]:points[i].y);
}var minVal=Math.min.apply(null,vals);
var maxVal=Math.max.apply(null,vals);

minVal-=range * 0.1;
maxVal+=range * 0.1;
range=maxVal-minVal;
ctx.strokeStyle="#222";
ctx.lineWidth=0.5;
for(var i=0;
i<=4;
i++){var yy=pad.top+chartH-(chartH / 4 * i);

ctx.lineTo(W-pad.right,yy);

for(var i=0;
i<points.length;
i++){var px=pad.left+(i /(points.length-1))* chartW;
var val=typeof points[i]==="number"?points[i]:points[i].y;

chartPoints.push({x:px,y:py});
}if(opts.fillColor){ctx.beginPath();

for(var i=1;
i<chartPoints.length;
i++)ctx.lineTo(chartPoints[i].x,chartPoints[i].y);
ctx.lineTo(chartPoints[chartPoints.length-1].x,pad.top+chartH);
ctx.lineTo(chartPoints[0].x,pad.top+chartH);
ctx.closePath();
ctx.fillStyle=opts.fillColor;
ctx.fill();
}ctx.beginPath();
ctx.moveTo(chartPoints[0].x,chartPoints[0].y);
for(var i=1;
i<chartPoints.length;
i++)ctx.lineTo(chartPoints[i].x,chartPoints[i].y);
ctx.strokeStyle=opts.color||"#4CAF50";
ctx.lineWidth=opts.lineWidth||1.5;
ctx.stroke();
if(opts.showDots!==false&&points.length<=100){for(var i=0;
i<chartPoints.length;
i++){ctx.beginPath();
ctx.arc(chartPoints[i].x,chartPoints[i].y,2,0,Math.PI * 2);
